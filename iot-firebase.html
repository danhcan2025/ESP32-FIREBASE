Dán code này vào dự án của bạn (Node.js / Browser). Mỗi thiết bị/trình duyệt kết nối Firebase sẽ nhận cập nhật theo thời gian thực. 

// ─── CÀI ĐẶT ───────────────────────────────────
// npm install firebase

import { initializeApp } from "firebase/app";
import { getDatabase, ref, set,
         onValue, serverTimestamp } from "firebase/database";

// ─── KHỞI TẠO ──────────────────────────────────
const app = initializeApp({
  apiKey:      "YOUR_API_KEY",
  databaseURL: "https://YOUR-PROJECT-rtdb.firebaseio.com",
  projectId:   "YOUR_PROJECT_ID",
});
const db = getDatabase(app);

// ─── CẤU TRÚC DATABASE ─────────────────────────
// iot-home/
//   switches/
//     den-phong-khach: { state: false, ... }
//     dieu-hoa:        { state: true,  ... }
//     quat:            { state: false, ... }
//     o-cam:           { state: true,  ... }

// ─── BẬT / TẮT THIẾT BỊ ───────────────────────
async function toggleDevice(deviceId, state) {
  const deviceRef = ref(db, `iot-home/switches/${deviceId}`);
  await set(deviceRef, {
    state:     state,          // true = BẬT, false = TẮT
    updatedAt: serverTimestamp(),
    updatedBy: "device-A",    // ID thiết bị gửi lệnh
  });
}

// ─── LẮNG NGHE THAY ĐỔI (REALTIME) ────────────
function watchAllSwitches() {
  const switchRef = ref(db, "iot-home/switches");

  onValue(switchRef, (snapshot) => {
    const data = snapshot.val();
    Object.entries(data || {}).forEach(([id, info]) => {
      console.log(`${id}: ${info.state ? "BẬT 🟢" : "TẮT ⚫"}`);
      updateHardware(id, info.state); // cập nhật GPIO/relay
    });
  });
}
