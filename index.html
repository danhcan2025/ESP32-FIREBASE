/*  */
#include 
#include

#define FIREBASE_PROJECT_ID "gemini-55ab8"
#define FIREBASE_API_KEY "AIzaSyBSJwS5xFjXAczjBmVx1gvhMCiZFqKHFYU"

// Chân GPIO
const int relayPins[] = {1, 2, 3, 4};       // Relay 1, 2, 3, 4
const int buttonPins[] = {5, 6, 7, 10};    // Button 1, 2, 3, 4
const int statusLed = 8;                   // LED trên board
const char* deviceIds[] = {"relay1", "relay2", "relay3", "relay4"};

bool relayStates[] = {false, false, false, false};
int lastBtnState[] = {HIGH, HIGH, HIGH, HIGH};
unsigned long lastDebounce[] = {0, 0, 0, 0};

FirebaseData fbdo;
FirebaseAuth auth;
FirebaseConfig config;
WiFiManager wm; // Global để có thể reset trong loop

void blinkLed() {
    digitalWrite(statusLed, HIGH); delay(100); digitalWrite(statusLed, LOW);
}

/*  */
void setup() {
    Serial.begin(115200);
    pinMode(statusLed, OUTPUT);
    for(int i=0; i<4; i++) {
        pinMode(relayPins[i], OUTPUT);
        digitalWrite(relayPins[i], LOW);
        pinMode(buttonPins[i], INPUT_PULLUP);
    }
    
    // Kết nối WiFi
    wm.autoConnect("ESP32-C3-Config", "12345678");
    WiFi.mode(WIFI_STA); 
    
    // Đồng bộ giờ chuẩn để Firebase xác thực Token
    configTime(7 * 3600, 0, "pool.ntp.org", "time.nist.gov");
    struct tm timeinfo;
    Serial.print("Syncing Time");
    while(!getLocalTime(&timeinfo)) { delay(500); Serial.print("."); }
    Serial.println("\nTime Synced!");

    // Cấu hình Firebase
    config.api_key = FIREBASE_API_KEY;
    Firebase.signUp(&config, &auth, "", "");
    Firebase.begin(&config, &auth);
    Firebase.reconnectWiFi(true);
    
    for(int i=0; i<3; i++) { blinkLed(); delay(100); }
    Serial.println("--- SYSTEM READY ---");
}

/*  */
void loop() {
    if (Firebase.ready()) {
        // 1. Gửi Heartbeat (Mỗi 15 giây) để Web báo Trực tuyến
        static unsigned long lastHb = 0;
        if (millis() - lastHb > 15000) {
            lastHb = millis();
            time_t now = time(nullptr);
            unsigned long long nowMs = (unsigned long long)now * 1000;
            String path = "artifacts/gemini-55ab8/public/data/system/status";
            String content = "{\"fields\":{\"last_seen\":{\"integerValue\":\"" + String(nowMs) + "\"}}}";
            Firebase.Firestore.patchDocument(&fbdo, FIREBASE_PROJECT_ID, "", path.c_str(), content.c_str(), "last_seen");
        }

        // 2. Kiểm tra lệnh Reset WiFi từ Dashboard Web
        static unsigned long lastCmdCheck = 0;
        if (millis() - lastCmdCheck > 5000) {
            lastCmdCheck = millis();
            String cmdPath = "artifacts/gemini-55ab8/public/data/system/commands";
            if (Firebase.Firestore.getDocument(&fbdo, FIREBASE_PROJECT_ID, "", cmdPath.c_str(), "")) {
                if (fbdo.payload().indexOf("\"reset_wifi\":{\"booleanValue\":true}") > 0) {
                    Serial.println(">>> RESET WIFI COMMAND RECEIVED!");
                    Firebase.Firestore.patchDocument(&fbdo, FIREBASE_PROJECT_ID, "", cmdPath.c_str(), "{\"fields\":{\"reset_wifi\":{\"booleanValue\":false}}}", "reset_wifi");
                    delay(1000); 
                    wm.resetSettings(); 
                    ESP.restart();
                }
            }
        }

        // 3. Xử lý Nút nhấn vật lý (GPIO 5, 6, 7, 10)
        for(int i=0; i<4; i++) {
            int reading = digitalRead(buttonPins[i]);
            if (reading != lastBtnState[i]) lastDebounce[i] = millis();
            if ((millis() - lastDebounce[i]) > 50) {
                if (reading == LOW && lastBtnState[i] == HIGH) {
                    relayStates[i] = !relayStates[i];
                    digitalWrite(relayPins[i], relayStates[i] ? HIGH : LOW);
                    String path = "artifacts/gemini-55ab8/public/data/devices/" + String(deviceIds[i]);
                    String content = "{\"fields\":{\"status\":{\"booleanValue\":" + String(relayStates[i] ? "true" : "false") + "}}}";
                    Firebase.Firestore.patchDocument(&fbdo, FIREBASE_PROJECT_ID, "", path.c_str(), content.c_str(), "status");
                    Serial.printf("Physical Button TB%d: %s\n", i+1, relayStates[i] ? "ON" : "OFF");
                    blinkLed();
                }
            }
            lastBtnState[i] = reading;
        }

        // 4. Đồng bộ lệnh từ Dashboard Web (Mỗi 2 giây)
        static unsigned long lastSync = 0;
        if(millis() - lastSync > 2000) {
            lastSync = millis();
            for(int i=0; i<4; i++) {
                String path = "artifacts/gemini-55ab8/public/data/devices/" + String(deviceIds[i]);
                if (Firebase.Firestore.getDocument(&fbdo, FIREBASE_PROJECT_ID, "", path.c_str(), "")) {
                    bool cloudState = fbdo.payload().indexOf("\"booleanValue\":true") > 0;
                    if(cloudState != relayStates[i]) {
                        relayStates[i] = cloudState;
                        digitalWrite(relayPins[i], relayStates[i] ? HIGH : LOW);
                        Serial.printf("Web Command TB%d: %s\n", i+1, relayStates[i] ? "ON" : "OFF");
                        blinkLed();
                    }
                }
            }
        }
    }
}
